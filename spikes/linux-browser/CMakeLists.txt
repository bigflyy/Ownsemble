cmake_minimum_required(VERSION 3.21)
project(ownsemble_generated_cef_probe C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory(${CEF_LIBCEF_DLL_WRAPPER_PATH} wrapper)
add_executable(ownsemble_cef_probe cef_probe.cc)
target_include_directories(ownsemble_cef_probe PRIVATE ${CEF_ROOT})
target_link_libraries(ownsemble_cef_probe PRIVATE libcef_dll_wrapper "${CEF_ROOT}/Release/libcef.so" pthread dl)
set_target_properties(ownsemble_cef_probe PROPERTIES BUILD_RPATH "${CEF_ROOT}/Release")
